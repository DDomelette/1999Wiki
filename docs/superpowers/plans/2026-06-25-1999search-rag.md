```python
"""统一配置加载: settings.yaml + .env 环境变量覆盖。"""
from __future__ import annotations

import os
from dataclasses import dataclass, field
from pathlib import Path
from typing import Any

import yaml
from dotenv import load_dotenv


@dataclass
class ObsidianCfg:
    vault_path: str


@dataclass
class OllamaCfg:
    base_url: str
    embedding_model: str


@dataclass
class LLMCfg:
    provider: str
    base_url: str
    model: str
    api_key: str


@dataclass
class RAGCfg:
    chunk_size: int
    chunk_overlap: int
    top_k: int


@dataclass
class ServerCfg:
    backend_port: int
    streamlit_port: int
    gradio_port: int
    frontend_delay_seconds: int


@dataclass
class PathsCfg:
    project_root: Path
    data_raw: Path
    data_processed: Path
    vectorstore: Path
    frontend_html: Path


@dataclass
class Config:
    obsidian: ObsidianCfg
    ollama: OllamaCfg
    llm: LLMCfg
    rag: RAGCfg
    server: ServerCfg
    paths: PathsCfg


_config: Config | None = None


def get_config() -> Config:
    """单例配置加载。首次调用读取 settings.yaml, 并用 DEEPSEEK_API_KEY 覆盖 api_key。"""
    global _config
    if _config is not None:
        return _config

    project_root = Path(__file__).resolve().parents[1]
    # 加载 .env (若存在)
    load_dotenv(project_root / ".env")

    with open(project_root / "config" / "settings.yaml", "r", encoding="utf-8") as f:
        raw: dict[str, Any] = yaml.safe_load(f)

    # api_key: env 优先覆盖 yaml 中的空值
    api_key = os.environ.get("DEEPSEEK_API_KEY") or raw["llm"].get("api_key", "") or ""

    cfg = Config(
        obsidian=ObsidianCfg(vault_path=raw["obsidian"]["vault_path"]),
        ollama=OllamaCfg(
            base_url=raw["ollama"]["base_url"],
            embedding_model=raw["ollama"]["embedding_model"],
        ),
        llm=LLMCfg(
            provider=raw["llm"]["provider"],
            base_url=raw["llm"]["base_url"],
            model=raw["llm"]["model"],
            api_key=api_key,
        ),
        rag=RAGCfg(
            chunk_size=raw["rag"]["chunk_size"],
            chunk_overlap=raw["rag"]["chunk_overlap"],
            top_k=raw["rag"]["top_k"],
        ),
        server=ServerCfg(
            backend_port=raw["server"]["backend_port"],
            streamlit_port=raw["server"]["streamlit_port"],
            gradio_port=raw["server"]["gradio_port"],
            frontend_delay_seconds=raw["server"]["frontend_delay_seconds"],
        ),
        paths=PathsCfg(
            project_root=project_root,
            data_raw=project_root / "data" / "raw",
            data_processed=project_root / "data" / "processed",
            vectorstore=project_root / "vectorstore",
            frontend_html=project_root / "frontend" / "html",
        ),
    )
    _config = cfg
    return cfg


def reset_config_for_test() -> None:
```

```python
"""Obsidian markdown 正文清洗: 去图片/wikilink/callout 标记/脚注/HTML 标签, 折叠空行。"""
from __future__ import annotations

import re


def clean_markdown(text: str) -> str:
    if not text:
        return ""

    # 1. 图片嵌入: ![[x.png]] 和 ![alt](url)
    text = re.sub(r"!\[\[[^\]]*\]\]", "", text)
    text = re.sub(r"!\[[^\]]*\]\([^)]*\)", "", text)

    # 2. wikilink: [[a|b]] -> b ; [[a]] -> a
    text = re.sub(r"\[\[([^\]|]+)\|([^\]]+)\]\]", r"\2", text)
    text = re.sub(r"\[\[([^\]]+)\]\]", r"\1", text)

    # 3. callout 标记行: > [!type]+/- 标题  (整行删除, 保留其后以 > 开头的内容行)
    text = re.sub(r"^>\s*\[![^\]]*\][+-]?.*$", "", text, flags=re.MULTILINE)
    # 去掉引用块行首的 "> " (保留内容)
    text = re.sub(r"^>\s?", "", text, flags=re.MULTILINE)

    # 4. 脚注引用标记 [^id]
    text = re.sub(r"\[\^[^\]]+\]", "", text)
    # 脚注定义块: 单独一行的 [^id]: xxx
    text = re.sub(r"^\[\^[^\]]+\]:.*$", "", text, flags=re.MULTILINE)

    # 5. HTML 标签 (保留标签内文本)
    text = re.sub(r"<[^>]+>", "", text)

    # 6. 折叠 3+ 连续换行为 2 个
    text = re.sub(r"\n{3,}", "\n\n", text)
    # 去除每行尾部空白
    text = "\n".join(line.rstrip() for line in text.splitlines())
    return text.strip()
```

```python
"""从 Obsidian vault 递归提取 .md, 解析 frontmatter, 清洗正文, 落盘 documents.jsonl 与镜像 raw。"""
from __future__ import annotations

import json
import shutil
from pathlib import Path
from typing import Any

import frontmatter

from config.config import Config
from src.utils.text_cleaner import clean_markdown

# 顶级目录前缀 -> category
_CATEGORY_MAP = [
    ("100-UTTU人物合辑", "人物"),
    ("200-相从心生", "心相"),
    ("300-以影像之", "剧情"),
    ("400-箱外世界", "世界"),
    ("500-箱外阵营", "阵营"),
    ("600-箱中日历", "日历"),
]

# 跳过的相对路径前缀 / 文件名
_EXCLUDE_PREFIXES = (
    ".obsidian",
    "assets",
    "000-箱的构造/templates",
    "000-箱的构造/script",
    "000-箱的构造/插件",
    "000-箱的构造/README",
)
_EXCLUDE_FILES = {"1999 Wiki&Note Obsidian Vault.md"}


def _category_for(rel_path: str) -> str | None:
    rel_norm = rel_path.replace("\\", "/")
    for prefix, cat in _CATEGORY_MAP:
        if rel_norm.startswith(prefix):
            return cat
    return None


def _is_kanban(text: str) -> bool:
    return "kanban-plugin" in text[:500]


def extract_file(md_path: Path, vault_root: Path, cfg: Config) -> dict | None:
    """处理单个 .md。返回文档 dict, 若应跳过返回 None。"""
    try:
        rel = md_path.relative_to(vault_root).as_posix()
    except ValueError:
        return None

    # 跳过规则
    if rel in _EXCLUDE_FILES:
        return None
    if any(rel.startswith(p) for p in _EXCLUDE_PREFIXES):
        return None
    category = _category_for(rel)
    if category is None:
        return None

    try:
        with open(md_path, "r", encoding="utf-8") as f:
            raw_text = f.read()
    except UnicodeDecodeError:
        with open(md_path, "r", encoding="utf-8", errors="replace") as f:
            raw_text = f.read()

    if _is_kanban(raw_text):
        return None

    post = frontmatter.loads(raw_text)
    meta: dict[str, Any] = dict(post.metadata)
    body = clean_markdown(post.content)
    if not body.strip():
        return None

    name = str(meta.get("Name") or md_path.stem)
    doc_id = rel[:-3]  # 去扩展名
    return {
        "id": doc_id,
        "source": rel,
        "name": name,
        "category": category,
        "metadata": meta,
        "text": body,
    }


def extract_all(cfg: Config) -> list[dict]:
    """提取整个 vault, 写入 documents.jsonl 与 raw 镜像。返回文档列表。"""
    vault_root = Path(cfg.obsidian.vault_path)
    if not vault_root.exists():
        raise FileNotFoundError(f"Vault not found: {vault_root}")

    raw_dir = cfg.paths.data_raw
    proc_dir = cfg.paths.data_processed
    raw_dir.mkdir(parents=True, exist_ok=True)
    proc_dir.mkdir(parents=True, exist_ok=True)

    docs: list[dict] = []
    for md_path in vault_root.rglob("*.md"):
        doc = extract_file(md_path, vault_root, cfg)
        if doc is None:
            continue
        docs.append(doc)
        # 镜像 raw
        rel = Path(doc["source"])
        dest = raw_dir / rel
        dest.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(md_path, dest)

    # 写 jsonl (覆盖)
    jsonl_path = proc_dir / "documents.jsonl"
    with open(jsonl_path, "w", encoding="utf-8") as f:
        for d in docs:
```

```python
"""Chroma 向量库: 构建/加载。"""
from __future__ import annotations

import json
from pathlib import Path

from langchain_chroma import Chroma
from langchain_core.documents import Document
from langchain_text_splitters import RecursiveCharacterTextSplitter

from config.config import Config
from src.rag.embeddings import get_embeddings


def _splitter(cfg: Config) -> RecursiveCharacterTextSplitter:
    return RecursiveCharacterTextSplitter(
        chunk_size=cfg.rag.chunk_size,
        chunk_overlap=cfg.rag.chunk_overlap,
        separators=["\n\n", "\n", "。", "；", "，", " ", ""],
    )


def _load_docs(cfg: Config) -> list[Document]:
    jsonl_path = cfg.paths.data_processed / "documents.jsonl"
    if not jsonl_path.exists():
        raise FileNotFoundError(f"documents.jsonl not found at {jsonl_path}; run scripts/extract_data.py first")
    docs: list[Document] = []
    with open(jsonl_path, "r", encoding="utf-8") as f:
        for line in f:
            d = json.loads(line)
            docs.append(Document(
                page_content=d["text"],
                metadata={
                    "source": d["source"],
                    "name": d["name"],
                    "category": d["category"],
                    "id": d["id"],
                },
            ))
    return docs


def build_vectorstore(cfg: Config, docs: list[dict] | None = None) -> Chroma:
    """从 documents.jsonl 构建并持久化 Chroma。docs 可传入直接构建 (用于测试)。"""
    embeddings = get_embeddings(cfg)
    persist_dir = cfg.paths.vectorstore
    persist_dir.mkdir(parents=True, exist_ok=True)

    if docs is None:
        raw_docs = _load_docs(cfg)
    else:
        raw_docs = [
            Document(page_content=d["text"],
                     metadata={"source": d["source"], "name": d["name"],
                               "category": d["category"], "id": d["id"]})
            for d in docs
        ]

    splitter = _splitter(cfg)
    chunks = splitter.split_documents(raw_docs)
    vectorstore = Chroma.from_documents(
        documents=chunks,
        embedding=embeddings,
        persist_directory=str(persist_dir),
        collection_name="reverse1999",
    )
    return vectorstore


def load_vectorstore(cfg: Config) -> Chroma:
    """加载已持久化的 Chroma。若不存在则自动构建。"""
    persist_dir = cfg.paths.vectorstore
    embeddings = get_embeddings(cfg)
    if not (persist_dir.exists() and any(persist_dir.iterdir())):
        return build_vectorstore(cfg)
    return Chroma(
        persist_directory=str(persist_dir),
        embedding_function=embeddings,
        collection_name="reverse1999",
    )
```

```python
"""检索器: 封装 Chroma 相似度搜索, 支持 category 过滤。"""
from __future__ import annotations

from typing import Any

from langchain_chroma import Chroma

from config.config import Config


class Retriever:
    def __init__(self, cfg: Config, vectorstore: Chroma) -> None:
        self._cfg = cfg
        self._vs = vectorstore

    def search(self, query: str, k: int | None = None, category: str | None = None) -> list[dict[str, Any]]:
        top_k = k or self._cfg.rag.top_k
        kwargs: dict[str, Any] = {"k": top_k}
        if category:
            kwargs["filter"] = {"category": category}
        try:
            results = self._vs.similarity_search_with_relevance_scores(query, **kwargs)
        except Exception:
            # 部分版本不支持 relevance_scores, 回退
            docs = self._vs.similarity_search(query, **kwargs)
            results = [(d, 0.0) for d in docs]
        out: list[dict[str, Any]] = []
        for doc, score in results:
            out.append({
                "name": doc.metadata.get("name", ""),
                "category": doc.metadata.get("category", ""),
                "source": doc.metadata.get("source", ""),
                "score": float(score),
                "content": doc.page_content,
            })
        return out
```

```python
"""RAG 链: retrieve -> prompt -> LLM, 返回 answer + sources。"""
from __future__ import annotations

from typing import Any

from langchain_openai import ChatOpenAI

from config.config import Config
from src.rag.prompts import get_rag_prompt
from src.rag.retriever import Retriever

_API_KEY_EMPTY_MSG = "请在 .env 中配置 DEEPSEEK_API_KEY 后再提问。"


class RAGChain:
    def __init__(self, cfg: Config, retriever: Retriever) -> None:
        self._cfg = cfg
        self._retriever = retriever
        self._llm = self._build_llm() if cfg.llm.api_key else None
        self._prompt = get_rag_prompt()

    def _build_llm(self) -> ChatOpenAI:
        return ChatOpenAI(
            base_url=self._cfg.llm.base_url,
            api_key=self._cfg.llm.api_key,
            model=self._cfg.llm.model,
            temperature=0.3,
        )

    def llm_ready(self) -> bool:
        return self._llm is not None

    def ask(self, question: str, category: str | None = None) -> dict[str, Any]:
        sources = self._retriever.search(question, category=category)
        if not self.llm_ready():
            return {"answer": _API_KEY_EMPTY_MSG, "sources": sources}

        if not sources:
            return {"answer": "知识库中未找到相关内容。", "sources": []}

        context = "\n\n".join(
            f"[{s['name']}] {s['content']}" for s in sources
        )
        messages = self._prompt.format_messages(context=context, question=question)
        try:
            resp = self._llm.invoke(messages)
            answer = resp.content if hasattr(resp, "content") else str(resp)
        except Exception as e:
            return {"answer": f"调用 LLM 失败: {e}", "sources": sources}
        return {"answer": answer, "sources": sources}
```

```python
    CORSMiddleware,
    allow_origins=[
        f"http://localhost:{cfg.server.streamlit_port}",
        f"http://localhost:{cfg.server.gradio_port}",
        f"http://127.0.0.1:{cfg.server.streamlit_port}",

    allow_headers=["*"],

# 启动时加载向量库与 RAG 链
_state: dict = {"vs": None, "retriever": None, "chain": None, "loaded": False}


def _ensure_loaded() -> None:
    if _state["loaded"]:
        return
    try:
        vs = load_vectorstore(cfg)
        retriever = Retriever(cfg, vs)
        chain = RAGChain(cfg, retriever)
        _state.update(vs=vs, retriever=retriever, chain=chain, loaded=True)
    except Exception as e:
        print(f"[backend] 加载失败: {e}")
        _state["loaded"] = False


@app.on_event("startup")
async def _startup() -> None:
    _ensure_loaded()


@app.get("/health", response_model=HealthResponse)
async def health() -> HealthResponse:
    _ensure_loaded()
    chain = _state.get("chain")
    vs = _state.get("vs")
    doc_count = 0
    try:
        if vs is not None:
            doc_count = vs._collection.count()
    except Exception:
        pass
    return HealthResponse(
        status="ok" if _state["loaded"] else "error",
        vectorstore_loaded=_state["loaded"],
        llm_ready=bool(chain and chain.llm_ready()),
        doc_count=doc_count,
    )


@app.post("/ask", response_model=AskResponse)
async def ask(req: AskRequest) -> JSONResponse:
    _ensure_loaded()
    chain = _state.get("chain")
    if chain is None:

            status_code=503,
        )

    sources = [
        SourceItem(name=s["name"], category=s["category"], source=s["source"], score=s["score"])
        for s in result.get("sources", [])
    ]
    return AskResponse(answer=result["answer"], sources=sources)


# 托管 HTML 前端 (在所有 API 路由之后挂载)
html_dir = cfg.paths.frontend_html

    app.mount("/", StaticFiles(directory=str(html_dir), html=True), name="html")
```

```javascript
    statusEl.textContent = '● 后端未连接';
    statusEl.style.color = '#ff6b6b';
  }

function addMsg(role, html) {
  const div = document.createElement('div');
  div.className = `msg ${role}`;
  div.innerHTML = html;
  messagesEl.appendChild(div);
  messagesEl.scrollTop = messagesEl.scrollHeight;

formEl.addEventListener('submit', async (e) => {
  e.preventDefault();
  const q = questionEl.value.trim();
  if (!q) return;
  addMsg('user', escapeHtml(q));
  questionEl.value = '';
  const typing = addMsg('bot', '<span class="typing">思考中…</span>');
  try {
    const res = await fetch('/ask', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ question: q, category: categoryEl.value || null }),
    });
    const data = await res.json();
    let html = `<div class="answer">${escapeHtml(data.answer)}</div>`;
    if (data.sources && data.sources.length) {
      const items = data.sources.map(s =>
        `<div class="source-item"><span class="name">${escapeHtml(s.name)}</span><span class="cat">${escapeHtml(s.category)}</span> · 相关度 ${s.score.toFixed(3)}</div>`
      ).join('');
      html += `<details class="sources"><summary>来源引用 (${data.sources.length})</summary><div class="source-list">${items}</div></details>`;
    }
    typing.innerHTML = html;
  } catch (err) {
    typing.innerHTML = `<div class="answer" style="color:#ff6b6b">请求失败: ${escapeHtml(String(err))}</div>`;
  }
});

function escapeHtml(s) {
  return String(s).replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
}
```

````markdown
## 模块职责

| 模块 | 职责 |
|------|------|
| `config/` | 统一加载 settings.yaml + .env,单例 Config |
| `src/utils/text_cleaner.py` | Obsidian markdown 正文清洗(纯函数) |
| `src/extraction/obsidian_extractor.py` | 递归遍历 vault、解析 frontmatter、分类、落盘 |

## 关键设计决策

1. **直读文件系统而非 Obsidian MCP**: MCP 连接不稳定,文件系统直读更可靠且无运行时依赖。
2. **Chroma 而非 FAISS**: 原生支持 metadata 过滤,适合 vault 分区检索。
3. **FastAPI 统一后端**: 三套前端共用同一 RAG 接口,避免重复实现;HTML 同源托管省一个进程。
4. **api_key 占位留空 + .env 覆盖**: 配置统一管理,不填也能启动(降级提示),便于交付。
5. **启动脚本健康检查 + 延迟启动前端**: 保证后端就绪前端才起,避免前端首屏报错。
````
